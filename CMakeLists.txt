cmake_minimum_required(VERSION 3.25)
include(cmake/read-version.cmake)
project(TriggerFishDrumFoundry VERSION ${DRUMFOUNDRY_VERSION} LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/VERSION.txt")
configure_file(cmake/version.hpp.in generated/drumfoundry/version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(DRUMFOUNDRY_BUILD_CLAP "Build the optional native CLAP instrument" OFF)
option(DRUMFOUNDRY_BUILD_STANDALONE "Build the native audio/MIDI test application" OFF)
option(DRUMFOUNDRY_BUILD_UI "Build the optional native Visage editor" OFF)
include(cmake/dependencies.cmake)
include(cmake/engine_sources.cmake)
add_library(drumfoundry_engine STATIC ${DRUMFOUNDRY_ENGINE_SOURCES})
target_include_directories(drumfoundry_engine PUBLIC engine engine/parameters)
target_include_directories(drumfoundry_engine PUBLIC "${PROJECT_BINARY_DIR}/generated")
target_link_libraries(drumfoundry_engine PUBLIC drumfoundry_dependencies)
include(cmake/editing.cmake)
# Host monitoring is opt-in and separate from raw synthesis/fitting.
add_library(drumfoundry_output STATIC engine/output/limiter.cpp)
target_include_directories(drumfoundry_output PUBLIC engine)
add_library(drumfoundry_native SHARED adapters/c_api/native.cpp)
target_link_libraries(drumfoundry_native PRIVATE drumfoundry_editing)
set_target_properties(drumfoundry_native PROPERTIES PREFIX "")
if(MINGW)
  target_link_options(drumfoundry_native PRIVATE -static -static-libgcc -static-libstdc++)
endif()
if(DRUMFOUNDRY_BUILD_UI)
  include(cmake/ui.cmake)
endif()
if(DRUMFOUNDRY_BUILD_CLAP OR DRUMFOUNDRY_BUILD_STANDALONE)
  include(cmake/clap.cmake)
endif()
if(DRUMFOUNDRY_BUILD_STANDALONE)
  include(cmake/standalone.cmake)
endif()
include(CTest)
include(cmake/windows-version.cmake)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()

# The CLAP preview adds a plugin; ordinary engine packages remain independent.
install(TARGETS drumfoundry_native RUNTIME DESTINATION lib LIBRARY DESTINATION lib)
install(FILES adapters/c_api/native.h DESTINATION include/drumfoundry)
install(DIRECTORY presets/factory/ DESTINATION presets/factory FILES_MATCHING PATTERN "*.fit.json")
install(FILES LICENSE.txt README.md DEVELOPMENT.md VERSION.txt CHANGELOG.md DESTINATION .)
install(FILES docs/manual.md DESTINATION docs)
install(FILES docs/images/drumfoundry.png DESTINATION docs/images)
install(FILES ${eigen_SOURCE_DIR}/COPYING.MPL2 DESTINATION licenses/eigen)
install(FILES ${json_SOURCE_DIR}/LICENSE.MIT DESTINATION licenses/json)
set(CPACK_PACKAGE_NAME "TriggerFish-DrumFoundry-engine")
if(DRUMFOUNDRY_BUILD_CLAP)
  set(CPACK_PACKAGE_NAME "TriggerFish-DrumFoundry-clap-preview")
endif()
if(DRUMFOUNDRY_BUILD_STANDALONE)
  set(CPACK_PACKAGE_NAME "TriggerFish-DrumFoundry-standalone-preview")
endif()
if(DRUMFOUNDRY_BUILD_UI)
  set(CPACK_PACKAGE_NAME "TriggerFish-DrumFoundry-ui-preview")
endif()
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR "ZIP")
include(CPack)
